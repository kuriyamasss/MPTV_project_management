const ProfileModal = ({ user, onClose, onSave }: { user: UserProfile, onClose: () => void, onSave: (u: any) => void }) => {
    const [formData, setFormData] = useState({ username: user.username, email: user.email, password: '', confirmPassword: '' });
    const { t } = useConfig();

    const handleSubmit = (e: React.FormEvent) => {
        e.preventDefault();
        if (formData.password && formData.password !== formData.confirmPassword) {
            alert("Passwords do not match");
            return;
        }
        onSave({ username: formData.username, email: formData.email });
    };

    return (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-[100] flex items-center justify-center">
            <div className="bg-white dark:bg-gray-800 p-6 rounded-xl w-full max-w-md shadow-2xl animate-in fade-in zoom-in duration-200">
                <div className="flex justify-between items-center mb-6">
                    <h2 className="text-xl font-bold flex items-center gap-2 text-gray-900 dark:text-gray-100">
                        <UserCog size={24} className="text-blue-600" /> {t.updateProfile}
                    </h2>
                    <button onClick={onClose}><X size={20} className="text-gray-400 hover:text-gray-600 dark:hover:text-gray-300"/></button>
                </div>
                <form onSubmit={handleSubmit} className="space-y-4">
                    <div>
                        <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">{t.username}</label>
                        <div className="relative">
                            <User className="absolute left-3 top-2.5 text-gray-400" size={16} />
                            <input className="w-full pl-9 pr-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" value={formData.username} onChange={e => setFormData({...formData, username: e.target.value})} required />
                        </div>
                    </div>
                    <div>
                        <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">{t.email}</label>
                        <div className="relative">
                            <Mail className="absolute left-3 top-2.5 text-gray-400" size={16} />
                            <input type="email" className="w-full pl-9 pr-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" value={formData.email} onChange={e => setFormData({...formData, email: e.target.value})} required />
                        </div>
                    </div>
                    <div className="pt-2 border-t border-gray-100 dark:border-gray-700">
                        <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">{t.newPassword}</label>
                        <div className="relative mb-3">
                            <Lock className="absolute left-3 top-2.5 text-gray-400" size={16} />
                            <input type="password" placeholder="New Password" className="w-full pl-9 pr-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" value={formData.password} onChange={e => setFormData({...formData, password: e.target.value})} />
                        </div>
                        <div className="relative">
                            <Lock className="absolute left-3 top-2.5 text-gray-400" size={16} />
                            <input type="password" placeholder="Confirm Password" className="w-full pl-9 pr-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" value={formData.confirmPassword} onChange={e => setFormData({...formData, confirmPassword: e.target.value})} />

const AppLayout = ({ children, user, logout, navigate, currentView, setCurrentView, onUpdateUser }: any) => {

  const [showProfileModal, setShowProfileModal] = useState(false);
  const { t } = useConfig();
  const dropdownRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target as Node)) {

const KanbanBoard = ({ project, setProject, onAddTask, onEditTask }: { project: Project, setProject: any, onAddTask: (colId: number) => void, onEditTask: (task: Task) => void }) => {

const GanttView = ({ project, setProject, onPreviewFile, onEditTask }: { project: Project, setProject: any, onPreviewFile: (file: string) => void, onEditTask: (task: Task) => void }) => {
  const [zoomLevel, setZoomLevel] = useState<"day" | "week" | "month">("day");
  const [expandedIds, setExpandedIds] = useState<Set<number>>(new Set([1001]));
  const [dragState, setDragState] = useState<any>(null);
  
  const [linkingTask, setLinkingTask] = useState<{ id: number, x: number, y: number } | null>(null);
  const [mousePos, setMousePos] = useState<{x:number, y:number} | null>(null);

  const [hoveredTask, setHoveredTask] = useState<{task: Task, x: number, y: number} | null>(null);
  const hoverTimeoutRef = useRef<any>(null);

  const tasks = useMemo(() => {
    const all = project.columns.flatMap(c => c.tasks).filter(t => t.start_date && t.end_date);
    all.sort((a, b) => new Date(a.start_date).getTime() - new Date(b.start_date).getTime());
    const map = new Map();
    all.forEach(t => map.set(t.id, { ...t, children: [], level: 0 }));
    const roots: any[] = [];
    all.forEach(t => {
      if (t.parent_id && map.has(t.parent_id)) {
        const p = map.get(t.parent_id);
        map.get(t.id).level = p.level + 1;
        p.children.push(map.get(t.id));
      } else {
        roots.push(map.get(t.id));
      }
    });
    const flat: any[] = [];
    const traverse = (nodes: any[]) => nodes.forEach(n => { flat.push(n); if (n.children.length) traverse(n.children); });
    traverse(roots);
    return flat;
  }, [project]);

  const visibleTasks = tasks.filter((t: any) => !t.parent_id || expandedIds.has(t.parent_id));

  const config = useMemo(() => {
    if (zoomLevel === "week") return { colWidth: 24, cols: 60, dayStep: 1, labelStep: 7 };
    if (zoomLevel === "month") return { colWidth: 12, cols: 90, dayStep: 1, labelStep: 30 };
    return { colWidth: 40, cols: 30, dayStep: 1, labelStep: 1 };
  }, [zoomLevel]);

  const minDate = new Date("2024-01-01"); 
  const days = Array.from({ length: config.cols }, (_, i) => { 
    const d = new Date(minDate); d.setDate(d.getDate() + i * config.dayStep); return d; 
  });

  const getOffset = (d: string) => {
    const date = new Date(d);
    if (isNaN(date.getTime())) return 0;
    return ((date.getTime() - minDate.getTime()) / (86400000)) * config.colWidth;
  };
  const getWidth = (s: string, e: string | undefined) => {
    const sDate = new Date(s);
    let eDate = e ? new Date(e) : new Date(s);
    if (isNaN(eDate.getTime())) eDate = new Date(s);
    if (isNaN(sDate.getTime()) || isNaN(eDate.getTime())) return 0;
    return Math.max(1, ((eDate.getTime() - sDate.getTime()) / 86400000) + 1) * config.colWidth;
  };
  const addDays = (d: string, n: number) => { 
    const x = new Date(d); 
    if (isNaN(x.getTime())) return new Date().toISOString();
    x.setDate(x.getDate() + n); 
    return x.toISOString(); 
  };
  const getDaysDiff = (start: string, end: string | undefined) => {
      const s = new Date(start);
      const e = end ? new Date(end) : new Date(start);
      if (isNaN(s.getTime()) || isNaN(e.getTime())) return 0;
      return Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)) + 1;
  };

  const handleMouseEnter = (task: Task, e: React.MouseEvent) => {
    if (hoverTimeoutRef.current) clearTimeout(hoverTimeoutRef.current);
    setHoveredTask({ task, x: e.clientX, y: e.clientY });
  };
  const handleMouseLeave = () => {
    hoverTimeoutRef.current = setTimeout(() => setHoveredTask(null), 300);
  };

  const updateTask = (id: number, updates: any) => {
    const np = { ...project };
    np.columns.forEach(c => {
      const t = c.tasks.find(x => x.id === id);
      if (t) Object.assign(t, updates);
    });
    setProject(np);
  };

  useEffect(() => {
    const move = (e: MouseEvent) => {
      if (linkingTask) {
        const svg = document.getElementById('gantt-svg');
        if (svg) {
            const rect = svg.getBoundingClientRect();
            setMousePos({ x: e.clientX - rect.left, y: e.clientY - rect.top });
        }
      }
      if (!dragState) return;
      const dx = Math.round((e.clientX - dragState.sx) / config.colWidth);
      if (dx === 0) return;
      if (dragState.type === "move") {
        updateTask(dragState.tid, { actual_start_date: addDays(dragState.os, dx), actual_end_date: addDays(dragState.oe, dx) });
      }
    };
    const up = () => {
        setDragState(null);
        setLinkingTask(null);
        setMousePos(null);
    };
    window.addEventListener("mousemove", move); 
    window.addEventListener("mouseup", up);
    return () => { window.removeEventListener("mousemove", move); window.removeEventListener("mouseup", up); }
  }, [dragState, config, linkingTask]);

  const startLinking = (e: React.MouseEvent, task: Task, type: 'start'|'end') => {
      e.stopPropagation();
      e.preventDefault();
      const svg = document.getElementById('gantt-svg');
      if (svg) {
        const rect = svg.getBoundingClientRect();
        const pl = getOffset(task.start_date);
        const pw = getWidth(task.start_date, task.end_date);
        const rowIndex = visibleTasks.findIndex(t => t.id === task.id);
        const y = (rowIndex * 48) + 24; 
        const x = type === 'end' ? pl + pw : pl; 
        setLinkingTask({ id: task.id, x, y });
        setMousePos({ x, y });
      }
  };

  const finishLinking = (e: React.MouseEvent, targetTask: Task) => {
      e.stopPropagation();
      if (linkingTask && linkingTask.id !== targetTask.id) {
          updateTask(linkingTask.id, { related_task_id: targetTask.id });
          setLinkingTask(null);
          setMousePos(null);
      }
  };

  const drawPath = (x1: number, y1: number, x2: number, y2: number) => {
      const midX = (x1 + x2) / 2;
      return `M ${x1} ${y1} C ${midX} ${y1}, ${midX} ${y2}, ${x2} ${y2}`;
  };

  return (
    <div className="bg-white dark:bg-gray-800 rounded-xl border border-gray-200 dark:border-gray-700 shadow-sm flex flex-col h-full overflow-hidden relative">
      {hoveredTask && !linkingTask && !dragState && (
         <div className="fixed z-[9999] bg-white dark:bg-gray-800 shadow-xl border border-gray-200 dark:border-gray-600 rounded-lg p-4 w-64 animate-in fade-in zoom-in duration-200 pointer-events-none" style={{ left: hoveredTask.x + 20, top: Math.min(hoveredTask.y, window.innerHeight - 200) }}>
            <h4 className="font-bold text-gray-900 dark:text-gray-100 mb-2">{hoveredTask.task.title}</h4>
            <div className="space-y-2 text-xs text-gray-600 dark:text-gray-400">
               <div className="flex items-center gap-2"><Avatar name={hoveredTask.task.assignee} /><span>{hoveredTask.task.assignee || "Unassigned"}</span></div>
               {hoveredTask.task.remarks && <div className="bg-gray-50 dark:bg-gray-700 p-2 rounded border border-gray-100 dark:border-gray-600 italic">{hoveredTask.task.remarks}</div>}
               {hoveredTask.task.attachments && hoveredTask.task.attachments.length > 0 && <div className="border-t border-gray-100 dark:border-gray-700 pt-2"><span className="font-semibold text-gray-700 dark:text-gray-300 block mb-1">Attachments:</span><ul className="space-y-1">{hoveredTask.task.attachments.map((file, i) => (<li key={i} className="flex items-center gap-2 text-blue-600 dark:text-blue-400"><Paperclip size={10}/> {file}</li>))}</ul></div>}
            </div>
         </div>
      )}

      <div className="px-4 py-2 border-b border-gray-200 dark:border-gray-700 bg-gray-50/50 dark:bg-gray-800/50 flex justify-between items-center">
        <div className="flex gap-4 text-xs font-medium text-gray-500 dark:text-gray-400">
          <div className="flex items-center gap-2"><span className="w-3 h-3 bg-gray-200 dark:bg-gray-600 rounded-sm border border-gray-300 dark:border-gray-500"></span> Planned</div>
          <div className="flex items-center gap-2"><span className="w-3 h-3 bg-blue-500 rounded-sm"></span> Actual</div>
        </div>
        <div className="flex bg-gray-100 dark:bg-gray-700 p-1 rounded-lg">
          {["day", "week", "month"].map((m: any) => (
            <button key={m} onClick={() => setZoomLevel(m)} className={`px-3 py-1 text-xs font-medium rounded-md capitalize transition-all ${zoomLevel===m ? "bg-white dark:bg-gray-600 text-gray-900 dark:text-white shadow-sm" : "text-gray-500 dark:text-gray-400 hover:text-gray-700 dark:hover:text-gray-200"}`}>{m}</button>
          ))}
        </div>
      </div>
      
      <div className="flex flex-1 overflow-hidden">
        <div className="w-64 border-r border-gray-200 dark:border-gray-700 flex flex-col bg-white dark:bg-gray-800 z-10 shadow-sm">
          <div className="h-10 border-b border-gray-100 dark:border-gray-700 flex items-center px-4 text-[10px] font-bold text-gray-400 uppercase tracking-wider bg-gray-50/30 dark:bg-gray-700/30">Task Name</div>
          <div className="overflow-y-hidden flex-1">
            {visibleTasks.map((t: any) => (
              <div key={t.id} className="h-12 flex items-center px-4 border-b border-gray-50 dark:border-gray-700 hover:bg-gray-50 dark:hover:bg-gray-700 transition-colors text-sm text-gray-700 dark:text-gray-200 group cursor-pointer" style={{ paddingLeft: `${16 + t.level * 16}px` }} onMouseEnter={(e) => handleMouseEnter(t, e)} onMouseLeave={handleMouseLeave} onClick={() => onEditTask(t)}>
                <button onClick={(e) => { e.stopPropagation(); const s = new Set(expandedIds); s.has(t.id)?s.delete(t.id):s.add(t.id); setExpandedIds(new Set(s)); }} className={`mr-2 p-0.5 rounded hover:bg-gray-200 dark:hover:bg-gray-600 text-gray-400 ${!t.children.length && "invisible"}`}>
                  {expandedIds.has(t.id)?<ChevronDown size={14}/>:<ChevronRight size={14}/>}
                </button>
                <span className="truncate group-hover:text-blue-600 dark:group-hover:text-blue-400 transition-colors">{t.title}</span>
              </div>
            ))}
          </div>
        </div>
        
        <div className="flex-1 overflow-auto relative bg-white dark:bg-gray-800">
          <div className="h-10 flex border-b border-gray-200 dark:border-gray-700 sticky top-0 z-20 bg-gray-50/90 dark:bg-gray-800/90 backdrop-blur w-max">
            {days.map((d, i) => (
              <div key={i} className="flex-shrink-0 border-r border-gray-100/50 dark:border-gray-700/50 flex flex-col justify-center items-center text-[10px] text-gray-400 select-none" style={{ width: config.colWidth }}>
                {i % config.labelStep === 0 && <span className="font-medium text-gray-600 dark:text-gray-400">{d.getDate()}</span>}
              </div>
            ))}
          </div>
          <div className="w-max relative pt-1" style={{ minHeight: visibleTasks.length * 48 }}>
            
            <svg id="gantt-svg" className="absolute inset-0 pointer-events-none z-0" style={{ width: '100%', height: '100%' }}>
                {visibleTasks.map((t:any, idx) => {
                    if (t.related_task_id) {
                        const target = visibleTasks.find((x:any) => x.id === t.related_task_id);
                        if (target) {
                            const tIdx = visibleTasks.findIndex((x:any) => x.id === target.id);
                            const startX = getOffset(t.start_date) + getWidth(t.start_date, t.end_date);
                            const startY = (idx * 48) + 24;
                            const endX = getOffset(target.start_date);
                            const endY = (tIdx * 48) + 24;
                            return <path key={`link-${t.id}`} d={drawPath(startX, startY, endX, endY)} stroke="#cbd5e1" strokeWidth="2" fill="none" markerEnd="url(#arrowhead)" />;
                        }
                    }
                    return null;
                })}
                {linkingTask && mousePos && (
                    <path d={drawPath(linkingTask.x, linkingTask.y, mousePos.x, mousePos.y)} stroke="#3b82f6" strokeWidth="2" fill="none" strokeDasharray="5,5" />
                )}
                <defs>
                    <marker id="arrowhead" markerWidth="10" markerHeight="7" refX="9" refY="3.5" orient="auto">
                        <polygon points="0 0, 10 3.5, 0 7" fill="#94a3b8" />
                    </marker>
                </defs>
            </svg>

            <div className="absolute inset-0 flex pointer-events-none">{days.map((_, i) => <div key={i} className="border-r border-gray-50 dark:border-gray-700 h-full" style={{ width: config.colWidth }}></div>)}</div>
            
            {visibleTasks.map((t: any) => {
                const pl = getOffset(t.start_date); const pw = getWidth(t.start_date, t.end_date);
                const al = t.actual_start_date ? getOffset(t.actual_start_date) : 0;
                const aw = t.actual_start_date ? getWidth(t.actual_start_date, t.actual_end_date) : 0;
                const duration = t.actual_start_date ? getDaysDiff(t.actual_start_date, t.actual_end_date) : 0;
                
                return (
                  <div key={t.id} className="h-12 relative border-b border-gray-50/50 dark:border-gray-700/50 hover:bg-gray-50/50 dark:hover:bg-gray-700/50 w-full group">
                    <div 
                        className="absolute top-2 h-7 bg-gray-100 dark:bg-gray-600 border border-gray-200 dark:border-gray-500 rounded flex items-center" 
                        style={{ left: pl, width: pw }}
                    >
                         <div 
                            onMouseDown={(e) => startLinking(e, t, 'end')}
                            className="absolute -right-2 top-1/2 -translate-y-1/2 w-3 h-3 bg-white dark:bg-gray-800 border-2 border-gray-300 dark:border-gray-500 rounded-full cursor-crosshair opacity-0 group-hover:opacity-100 hover:scale-125 hover:border-blue-500 z-20"
                         />
                         <div 
                            onMouseUp={(e) => finishLinking(e, t)}
                            className="absolute -left-3 top-0 bottom-0 w-4 z-20 cursor-pointer"
                         />
                    </div>
                    {t.actual_start_date && (
                      <div 
                        onMouseDown={(e) => setDragState({tid: t.id, type:"move", sx:e.clientX, os:t.actual_start_date, oe:t.actual_end_date})}
                        className={`absolute top-4 h-3 rounded shadow-sm text-[9px] text-white flex items-center justify-center cursor-grab active:cursor-grabbing hover:brightness-110 transition-all z-10 ${t.progress===100?"bg-emerald-500":"bg-blue-600"}`}
                        style={{ left: al, width: Math.max(aw, 8) }}
                      >
                        {aw > 40 && `${duration}d`}
                      </div>
                    )}
                  </div>
                );
            })}
          </div>
        </div>
      </div>
    </div>
  );
};

                        <span className="flex items-center gap-1 bg-gray-100 dark:bg-gray-700 px-2 py-1 rounded text-xs text-gray-600 dark:text-gray-300 border border-gray-200 dark:border-gray-600">
                           <Paperclip size={10} className="text-gray-400"/> {t.attachments.length}
                        </span>
                      ) : <span className="text-xs text-gray-300">-</span>}
                      <button className="text-gray-400 hover:text-blue-600 dark:hover:text-blue-400 transition-colors p-1 rounded hover:bg-blue-50 dark:hover:bg-blue-900/30"><Upload size={14} /></button>
                   </div>
                </td>
                <td className="px-4 py-2"><input type="text" className="w-full bg-transparent border-0 placeholder-gray-300 dark:placeholder-gray-600 focus:ring-0 p-0 text-sm truncate text-gray-700 dark:text-gray-300" placeholder="..." value={t.remarks || ""} onChange={(e) => update(t.id, "remarks", e.target.value)} /></td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

const ProjectFilesPage = ({ projects, onPreviewFile }: { projects: Project[], onPreviewFile: (file: string) => void }) => {

  const [view, setView] = useState<'grid' | 'list'>('grid');
  const [search, setSearch] = useState('');

  const filteredProjects = useMemo(() => {
    return projects.filter(p => p.name.toLowerCase().includes(search.toLowerCase()));
  }, [projects, search]);

    const start = (page - 1) * ITEMS_PER_PAGE;
    return filteredProjects.slice(start, start + ITEMS_PER_PAGE);
  }, [filteredProjects, page]);

           id: `${t.id}-${file}`,
           name: file,

        }))
      )
    );
  }, [openedProject]);

    const newSet = new Set(selectedFiles);
    newSet.has(id) ? newSet.delete(id) : newSet.add(id);

  };

  const handleBatchDownload = () => {
    alert(`Downloading ${selectedFiles.size} files... (Mock)`);

  };
  
  const handleUpload = () => {
      alert("Upload functionality would open a file picker here.");
  };

                <button onClick={() => setView("list")} className={`p-2 rounded-md transition-all ${view === "list" ? "bg-gray-100 dark:bg-gray-700 text-gray-900 dark:text-white" : "text-gray-400 hover:text-gray-600 dark:hover:text-gray-300"}`}><List size={18} /></button>

                   <p className="text-xs text-gray-400 text-center">{new Date(p.created_at).getFullYear()}</p>
                </div>
              ))}
            </div>
          ) : (
            <div className="bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl overflow-hidden shadow-sm">
               {paginatedProjects.map(p => (

           <div className="flex items-center gap-2 text-sm text-gray-500 dark:text-gray-400 mb-1">
              <span onClick={() => setOpenedProjectId(null)} className="cursor-pointer hover:underline">{t.allProjects}</span> / <span>{openedProject.name}</span>
           </div>
           <h1 className="text-3xl font-bold text-gray-900 dark:text-gray-100 tracking-tight">{t.files}</h1>
        </div>
        <div className="flex-1 text-right flex justify-end gap-3">
           <button onClick={handleUpload} className="bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 text-gray-700 dark:text-gray-200 hover:bg-gray-50 dark:hover:bg-gray-700 px-4 py-2.5 rounded-lg font-medium inline-flex items-center gap-2 shadow-sm transition-colors">

           >
             <Download size={18} /> {t.downloadSelected} ({selectedFiles.size})
           </button>
        </div>
      </div>

      <div className="bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl overflow-hidden shadow-sm">
        <table className="w-full text-sm text-left">

                  <td className="px-6 py-4"><input type="checkbox" checked={selectedFiles.has(file.id)} onChange={() => toggleSelect(file.id)} className="rounded border-gray-300 text-blue-600 focus:ring-blue-500 dark:bg-gray-700 dark:border-gray-600" /></td>
                  <td className="px-6 py-4 font-medium text-gray-900 dark:text-gray-200 flex items-center gap-2">
                    <FileDigit size={16} className="text-blue-500" /> {file.name}

                      <button onClick={() => onPreviewFile(file.name)} className="p-1.5 text-gray-400 hover:text-blue-600 hover:bg-blue-50 dark:hover:bg-blue-900/30 rounded" title="Preview"><Eye size={16}/></button>

                  </td>

// 5.1 Project Board Page
const ProjectBoardPage = ({ project, setProject, goBack, onPreviewFile }: { project: Project, setProject: any, goBack: () => void, onPreviewFile: (f:string) => void }) => {
  const [view, setView] = useState<'kanban'|'gantt'|'table'>('kanban');
  const [showCreateModal, setShowCreateModal] = useState(false);
  const [targetColumnId, setTargetColumnId] = useState<number | null>(null);
  const [newTask, setNewTask] = useState({ title: "", description: "", priority: "medium", start_date: "", end_date: "", actual_start_date: "", actual_end_date: "", parent_id: "", related_task_id: "" });
  const [editingTask, setEditingTask] = useState<Task | null>(null);
  const [showSettingsModal, setShowSettingsModal] = useState(false);
  const [projectSettings, setProjectSettings] = useState({ name: project.name, description: project.description, status: project.status });
  const { t } = useConfig();
  const allTasks = useMemo(() => project.columns.flatMap(c => c.tasks), [project]);

  const handleOpenCreateModal = (colId?: number) => {
    setTargetColumnId(colId || project.columns[0].id);
    setNewTask({ title: "", description: "", priority: "medium", start_date: "", end_date: "", actual_start_date: "", actual_end_date: "", parent_id: "", related_task_id: "" });
    setShowCreateModal(true);
  };
  const create = (e: React.FormEvent) => {
    e.preventDefault();
    if (!targetColumnId) return;
    const newProject = { ...project };
    const col = newProject.columns.find(c => c.id === targetColumnId);
    if (col) {
      col.tasks.push({
        id: Date.now(),
        ...newTask,
        priority: newTask.priority as Priority,
        progress: 0,
        parent_id: newTask.parent_id ? parseInt(newTask.parent_id) : undefined,
        related_task_id: newTask.related_task_id ? parseInt(newTask.related_task_id) : undefined,
        attachments: []
      } as Task);
      setProject(newProject);
    }
    setShowCreateModal(false);
  };
  const handleSaveEdit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!editingTask) return;
    const newProject = { ...project };
    newProject.columns.forEach(col => {
      const idx = col.tasks.findIndex(t => t.id === editingTask.id);
      if (idx > -1) col.tasks[idx] = { ...editingTask };
    });
    setProject(newProject);
    setEditingTask(null);
  };
  const handleDeleteTask = () => {
    if (!editingTask) return;
    if (!confirm("Are you sure you want to delete this task?")) return;
    const newProject = { ...project };
    newProject.columns.forEach(col => {
      col.tasks = col.tasks.filter(t => t.id !== editingTask.id);
    });
    setProject(newProject);
    setEditingTask(null);
  };
  const handleSaveProjectSettings = (e: React.FormEvent) => {
      e.preventDefault();
      const updatedProject = { ...project, ...projectSettings };
      setProject(updatedProject);
      setShowSettingsModal(false);
      if (projectSettings.status === 'inactive') {
          goBack();
      }
  };

  return (
    <div className="flex flex-col h-full">
      <div className="flex justify-between items-center mb-6 flex-shrink-0">
        <div>
          <div className="flex items-center gap-2 text-xs font-medium text-gray-400 mb-1 uppercase tracking-wide">
            <button onClick={goBack} className="hover:text-blue-600 transition-colors">{t.projects}</button> / <span>{project.name}</span>
          </div>
          <h1 className="text-2xl font-bold text-gray-900 dark:text-gray-100 tracking-tight">{project.name}</h1>
        </div>
        <div className="flex items-center gap-4">
          <button onClick={() => { setProjectSettings({ name: project.name, description: project.description, status: project.status }); setShowSettingsModal(true); }} className="p-2 rounded-md text-gray-400 hover:text-gray-600 dark:hover:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors" title="Project Settings"><Settings size={20} /></button>
          <div className="bg-gray-100 dark:bg-gray-800 p-1 rounded-lg flex border border-gray-200 dark:border-gray-700">
            {[ { id: 'kanban', icon: Kanban, label: 'Board' }, { id: 'gantt', icon: Calendar, label: 'Gantt' }, { id: 'table', icon: TableIcon, label: 'Table' } ].map(v => (
              <button key={v.id} onClick={() => setView(v.id as any)} className={`flex items-center gap-2 px-3 py-1.5 rounded-md text-sm font-medium transition-all ${view === v.id ? "bg-white dark:bg-gray-600 text-gray-900 dark:text-white shadow-sm" : "text-gray-500 dark:text-gray-400 hover:text-gray-700 dark:hover:text-gray-200"}`}><v.icon size={16} /> {v.label}</button>
            ))}
          </div>
          <button onClick={() => handleOpenCreateModal()} className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg text-sm font-medium flex items-center gap-2 shadow-lg shadow-blue-500/20 transition-all hover:scale-105 active:scale-95"><Plus size={16} /> {t.createTask}</button>
        </div>
      </div>
      <div className="flex-1 min-h-0">
        {view === 'kanban' && <KanbanBoard project={project} setProject={setProject} onAddTask={handleOpenCreateModal} onEditTask={setEditingTask} />}
        {view === 'gantt' && <GanttView project={project} setProject={setProject} onPreviewFile={onPreviewFile} onEditTask={setEditingTask} />}
        {view === 'table' && <TableView project={project} setProject={setProject} />}
      </div>
      {showSettingsModal && (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-50 flex items-center justify-center">
            <div className="bg-white dark:bg-gray-800 p-6 rounded-xl w-full max-w-md shadow-2xl animate-in fade-in zoom-in duration-200">
                <div className="flex justify-between items-center mb-4"><h2 className="text-xl font-bold flex items-center gap-2 text-gray-900 dark:text-gray-100"><Settings size={20}/> {t.settings}</h2><button onClick={() => setShowSettingsModal(false)}><X size={20} className="text-gray-400"/></button></div>
                <form onSubmit={handleSaveProjectSettings} className="space-y-4">
                    <div><label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">Project Name</label><input className="w-full border border-gray-300 dark:border-gray-600 rounded-lg px-3 py-2 bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" value={projectSettings.name} onChange={e => setProjectSettings({...projectSettings, name: e.target.value})} required /></div>
                    <div><label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">Description</label><textarea className="w-full border border-gray-300 dark:border-gray-600 rounded-lg px-3 py-2 bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" rows={3} value={projectSettings.description} onChange={e => setProjectSettings({...projectSettings, description: e.target.value})} /></div>
                    <div><label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">Status</label><select className="w-full border border-gray-300 dark:border-gray-600 rounded-lg px-3 py-2 bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100" value={projectSettings.status} onChange={e => setProjectSettings({...projectSettings, status: e.target.value as any})}><option value="active">{t.active}</option><option value="inactive">{t.inactive}</option></select><p className="text-xs text-gray-500 dark:text-gray-400 mt-1">Inactive projects are moved to Archives and hidden from Dashboard.</p></div>
                    <div className="flex justify-end gap-2 pt-4"><button type="button" onClick={() => setShowSettingsModal(false)} className="px-4 py-2 text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-700 rounded-lg">{t.cancel}</button><button type="submit" className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700">{t.save}</button></div>
                </form>
            </div>
        </div>
      )}
      {showCreateModal && (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-50 flex items-center justify-center">
          <div className="bg-white dark:bg-gray-800 p-6 rounded-xl w-full max-w-lg shadow-2xl animate-in fade-in zoom-in duration-200 max-h-[90vh] overflow-y-auto">
            <div className="flex justify-between items-center mb-4"><h2 className="text-xl font-bold text-gray-900 dark:text-gray-100">{t.createTask}</h2><button onClick={() => setShowCreateModal(false)} className="text-gray-400 hover:text-gray-600"><X size={20}/></button></div>
            <form onSubmit={create} className="space-y-4">
              <input className="w-full border border-gray-300 dark:border-gray-600 rounded-lg px-4 py-2 bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Task Title" value={newTask.title} onChange={e=>setNewTask({...newTask, title:e.target.value})} required />
              <div className="grid grid-cols-2 gap-4">
                 <div><label className="block text-xs font-medium text-gray-500 dark:text-gray-400 mb-1">{t.priority}</label><select className="w-full border border-gray-300 dark:border-gray-600 rounded-lg px-3 py-2 bg-white dark:bg-gray-700 text-gray-900 dark:text-gray-100 text-sm" value={newTask.priority} onChange={e=>setNewTask({...newTask, priority:e.target.value})}><option value="high">High</option><option value="medium">Medium</option><option value="low">Low</option></select></div>

  const [newProject, setNewProject] = useState({ name: "", description: "" });
  const [search, setSearch] = useState('');
  const [page, setPage] = useState(1);
  const { t } = useConfig();
  const ITEMS_PER_PAGE = 10;
  
  const [showReportModal, setShowReportModal] = useState(false);
  const [reportConfig, setReportConfig] = useState<{ start: string, end: string, selected: Set<number> }>({ start: '', end: '', selected: new Set() });

  const getDefaultDates = () => {
    const today = new Date();
    const thisThursday = new Date(today);
    thisThursday.setDate(today.getDate() + (4 + 7 - today.getDay()) % 7);
    if (thisThursday > today) thisThursday.setDate(thisThursday.getDate() - 7);
    const lastThursday = new Date(thisThursday);
    lastThursday.setDate(thisThursday.getDate() - 7);
    return { start: lastThursday.toISOString().split('T')[0], end: thisThursday.toISOString().split('T')[0] };
  };

  const openReportModal = () => { setReportConfig({ ...getDefaultDates(), selected: new Set() }); setShowReportModal(true); };
  const handleInvertSelection = () => {
      const activeProjects = projects.filter(p => p.status === 'active');
      const allIds = activeProjects.map(p => p.id);
      const newSelected = new Set(reportConfig.selected);
      allIds.forEach(id => { if (newSelected.has(id)) { newSelected.delete(id); } else { newSelected.add(id); } });
      setReportConfig({ ...reportConfig, selected: newSelected });
  };
  const handleGenerateReport = () => { alert(`Generating report for ${reportConfig.selected.size} projects...`); setShowReportModal(false); };

  const filteredProjects = useMemo(() => projects.filter(p => p.status === 'active' && p.name.toLowerCase().includes(search.toLowerCase())), [projects, search]);

          <div className="relative"><Search className="absolute left-3 top-2.5 text-gray-400" size={16} /><input type="text" placeholder={t.search} className="pl-9 pr-4 py-2 border border-gray-200 dark:border-gray-700 rounded-lg text-sm focus:ring-2 focus:ring-blue-500 outline-none w-64 bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100" value={search} onChange={(e) => { setSearch(e.target.value); setPage(1); }} /></div>

"bg-gray-100 dark:bg-gray-700 text-gray-900 dark:text-white" : "text-gray-400 hover:text-gray-600 dark:hover:text-gray-300"}`}><LayoutGrid size={18} /></button><button onClick={() => setView("list")} className={`p-2 rounded-md transition-all ${view === "list" ? "bg-gray-100 dark:bg-gray-700 text-gray-900 dark:text-white" : "text-gray-400 hover:text-gray-600 dark:hover:text-gray-300"}`}><List size={18} /></button></div>

        </div>
      </div>
      {filteredProjects.length === 0 ? <div className="text-center py-20 text-gray-400">No active projects found.</div> : (
        view === "grid" ? (
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">{paginatedProjects.map(p => (<div key={p.id} onClick={() => openProject(p.id)} className="bg-white dark:bg-gray-800 p-6 rounded-xl border border-gray-200 dark:border-gray-700 hover:border-blue-400 dark:hover:border-blue-500 hover:shadow-xl hover:shadow-blue-500/5 transition-all cursor-pointer group relative overflow-hidden"><div className="absolute top-0 right-0 p-4 opacity-0 group-hover:opacity-100 transition-opacity text-blue-500 translate-x-2 group-hover:translate-x-0"><ChevronRight /></div><div className="w-12 h-12 bg-blue-50 dark:bg-blue-900/20 rounded-xl flex items-center justify-center text-blue-600 dark:text-blue-400 mb-4 group-hover:scale-110 transition-transform"><Briefcase size={24} /></div><h3 className="text-lg font-bold text-gray-900 dark:text-gray-100 mb-2">{p.name}</h3><p className="text-gray-500 dark:text-gray-400 text-sm line-clamp-2 h-10">{p.description}</p><div className="mt-4 pt-4 border-t border-gray-100 dark:border-gray-700 flex justify-between items-center text-xs text-gray-400 font-mono"><span>{new Date(p.created_at).toLocaleDateString()}</span></div></div>))}</div>

                  <div className="mt-4 pt-4 border-t border-gray-200 dark:border-gray-700 flex justify-between items-center text-xs text-gray-400 font-mono">
                    <span>{new Date(p.created_at).toLocaleDateString()}</span>
                  </div>
                </div>
              ))}
            </div>
          ) : (
            <div className="bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl overflow-hidden shadow-sm">
              {paginatedProjects.map(p => (
                <div key={p.id} onClick={() => openProject(p.id)} className="flex items-center p-4 border-b border-gray-100 dark:border-gray-700 last:border-0 hover:bg-gray-50 dark:hover:bg-gray-700 cursor-pointer transition-colors group grayscale hover:grayscale-0">
                  <div className="w-10 h-10 bg-gray-100 dark:bg-gray-700 rounded-lg flex items-center justify-center text-gray-500 dark:text-gray-400 mr-4"><Briefcase size={20} /></div>
                  <div className="flex-1">
                    <h3 className="font-semibold text-gray-700 dark:text-gray-300">{p.name}</h3>
                    <p className="text-sm text-gray-500 dark:text-gray-400">{p.description}</p>
                  </div>
                  <div className="text-sm text-gray-400 font-mono mr-8">{new Date(p.created_at).toLocaleDateString()}</div>
                  <ChevronRight className="text-gray-300 dark:text-gray-600 group-hover:text-gray-500" size={20} />
                </div>
              ))}
            </div>
          )
        )}
  
        {totalPages > 1 && (
          <div className="flex justify-center mt-8 gap-2">
             <button disabled={page===1} onClick={() => setPage(p => p-1)} className="px-3 py-1 rounded border dark:border-gray-700 disabled:opacity-50 hover:bg-gray-50 dark:hover:bg-gray-700 text-sm text-gray-600 dark:text-gray-400">Prev</button>
             <span className="px-3 py-1 text-sm text-gray-600 dark:text-gray-400">Page {page} of {totalPages}</span>
             <button disabled={page===totalPages} onClick={() => setPage(p => p+1)} className="px-3 py-1 rounded border dark:border-gray-700 disabled:opacity-50 hover:bg-gray-50 dark:hover:bg-gray-700 text-sm text-gray-600 dark:text-gray-400">Next</button>
          </div>
        )}
      </div>
    );
  };

// --- 6. 主应用 (App) ---
export default function App() {
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [currentView, setCurrentView] = useState<'dashboard' | 'files' | 'project' | 'archives'>('dashboard');
  const [selectedProjectId, setSelectedProjectId] = useState<number | null>(null);
  const [projects, setProjects] = useState<Project[]>(INITIAL_DATA);
  const [previewFile, setPreviewFile] = useState<string | null>(null);
  const [lang, setLang] = useState<Language>('zh');
  const [theme, setTheme] = useState<Theme>('light');
  
  // App-level user state
  const [currentUser, setCurrentUser] = useState<UserProfile>({ username: 'Administrator', email: 'admin@example.com' });

  // Handle dark mode class on body
  useEffect(() => {
    if (theme === 'dark') {
      document.documentElement.classList.add('dark');
    } else {
      document.documentElement.classList.remove('dark');
    }
  }, [theme]);

  // Mock SetProject to simulate state updates
  const handleSetProject = (newProject: Project) => {
    const updatedProjects = projects.map(p => p.id === newProject.id ? newProject : p);
    setProjects(updatedProjects);
  };
  
  const handleUpdateUser = (updatedData: any) => {
      setCurrentUser(prev => ({ ...prev, ...updatedData }));
  };

  const handleCreateProject = (projectData: any) => {
     const newProject: Project = {
       id: Date.now(),
       name: projectData.name,
       description: projectData.description,
       status: 'active',
       created_at: new Date().toISOString(),
       columns: [
         { id: Date.now() + 1, name: 'To Do', tasks: [] },
         { id: Date.now() + 2, name: 'In Progress', tasks: [] },
         { id: Date.now() + 3, name: 'Done', tasks: [] },
         { id: Date.now() + 4, name: "Won't Do", tasks: [] }
       ]
     };
     setProjects([...projects, newProject]);
  };

  const handleLogout = () => {
    setIsLoggedIn(false);
    setCurrentView('dashboard');
  };

  // Provide Config Context
  const configValue = {
    lang, setLang,
    theme, setTheme,
    t: TRANSLATIONS[lang]
  };

  return (
    <ConfigContext.Provider value={configValue}>
      {!isLoggedIn ? (
        <LoginPage onLogin={() => setIsLoggedIn(true)} />
      ) : (
        <AppLayout 
          user={currentUser} 
          logout={handleLogout} 
          navigate={(view: any) => {
             if (view === '/') setCurrentView('dashboard');
             else console.log('nav', view);
          }}
          currentView={currentView}
          setCurrentView={setCurrentView}
          onUpdateUser={handleUpdateUser}
        >
          {currentView === 'dashboard' && (
            <DashboardPage 
              projects={projects} 
              openProject={(id) => { setSelectedProjectId(id); setCurrentView('project'); }} 
              onCreateProject={handleCreateProject}
            />
          )}

          {currentView === 'archives' && (
            <ArchivesPage 
              projects={projects} 
              openProject={(id) => { setSelectedProjectId(id); setCurrentView('project'); }} 
            />
          )}

          {currentView === 'files' && (
            <ProjectFilesPage projects={projects} onPreviewFile={setPreviewFile} />
          )}
          
          {currentView === 'project' && selectedProjectId && (
            <ProjectBoardPage 
              project={projects.find(p => p.id === selectedProjectId)!} 
              setProject={handleSetProject} 
              goBack={() => setCurrentView(projects.find(p => p.id === selectedProjectId)?.status === 'active' ? 'dashboard' : 'archives')}
              onPreviewFile={setPreviewFile}
            />
          )}

          <FilePreviewModal file={previewFile} onClose={() => setPreviewFile(null)} />
        </AppLayout>
      )}
    </ConfigContext.Provider>
  );
}
